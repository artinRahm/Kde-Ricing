Arch Linux KDE Plasma Ricing Guide

Complete walkthrough from base install to a fully riced KDE Plasma desktop on Arch Linux

~45 min read Intermediate Arch Linux

1. Base System Installation

Start with the official Arch Linux installation medium. Boot into the live environment and follow these steps:

1.1 Verify Boot Mode

# Check if booted in UEFI mode
ls /sys/firmware/efi/efivars

If the directory exists, you're in UEFI mode. Otherwise, you're in BIOS mode.

1.2 Connect to Internet

# For WiFi
iwctl
# Inside iwctl:
station wlan0 connect "SSID"

# For Ethernet (usually auto)
ping archlinux.org

1.3 Partition Disks

# List disks
lsblk

# Partition with fdisk (UEFI example)
fdisk /dev/nvme0n1
# Create: EFI (512M, type EFI), Root (rest, type Linux filesystem)

1.4 Format & Mount

# Format partitions
mkfs.fat -F32 /dev/nvme0n1p1  # EFI
mkfs.ext4 /dev/nvme0n1p2      # Root

# Mount
mount /dev/nvme0n1p2 /mnt
mkdir -p /mnt/boot/efi
mount /dev/nvme0n1p1 /mnt/boot/efi

1.5 Install Base System

# Install base packages + essentials
pacstrap -K /mnt base linux linux-firmware \
    base-devel networkmanager sudo vim git \
    intel-ucode  # or amd-ucode

# Generate fstab
genfstab -U /mnt >> /mnt/etc/fstab

1.6 Chroot & Configure

arch-chroot /mnt

# Timezone
ln -sf /usr/share/zoneinfo/Region/City /etc/localtime
hwclock --systohc

# Localization
echo "en_US.UTF-8 UTF-8" >> /etc/locale.gen
locale-gen
echo "LANG=en_US.UTF-8" > /etc/locale.conf

# Hostname
echo "archrice" > /etc/hostname

# Root password
passwd

# Create user
useradd -m -G wheel -s /bin/bash youruser
passwd youruser
EDITOR=vim visudo  # Uncomment %wheel ALL=(ALL:ALL) ALL

1.7 Bootloader (systemd-boot for UEFI)

bootctl install

# Create loader entry
cat > /boot/loader/entries/arch.conf << 'EOF'
title   Arch Linux
linux   /vmlinuz-linux
initrd  /intel-ucode.img
initrd  /initramfs-linux.img
options root=PARTUUID=$(blkid -s PARTUUID -o value /dev/nvme0n1p2) rw
EOF

# Configure loader
cat > /boot/loader/loader.conf << 'EOF'
default arch
timeout 3
console-mode keep
EOF

1.8 Enable Services & Reboot

systemctl enable NetworkManager
exit
umount -R /mnt
reboot

2. Post-Installation Setup

2.1 System Update & Essential Packages

sudo pacman -Syu

# Essential utilities
sudo pacman -S --needed \
    base-devel git curl wget unzip zip \
    htop btop neofetch fastfetch \
    man-db man-pages texinfo \
    reflector rsync openssh

2.2 Configure Pacman & Mirrors

# Enable parallel downloads & color
sudo sed -i 's/^#ParallelDownloads = 5/ParallelDownloads = 10/' /etc/pacman.conf
sudo sed -i 's/^#Color/Color/' /etc/pacman.conf
sudo sed -i 's/^#VerbosePkgLists/VerbosePkgLists/' /etc/pacman.conf

# Add ILoveCandy for fun
sudo sed -i '/^Color/a ILoveCandy' /etc/pacman.conf

# Update mirrors
sudo reflector --country "United States" --age 12 --protocol https --sort rate --save /etc/pacman.d/mirrorlist

2.3 AUR Helper (yay)

cd /tmp
git clone https://aur.archlinux.org/yay.git
cd yay
makepkg -si --noconfirm
cd ~ && rm -rf /tmp/yay

2.4 Enable Multilib (for Steam, Wine, etc.)

sudo sed -i '/^#\[multilib\]/,/^#Include/ s/^#//' /etc/pacman.conf
sudo pacman -Syu

3. KDE Plasma Installation

3.1 Minimal vs Full Plasma

Minimal (Recommended for Ricing)

sudo pacman -S plasma-meta konsole dolphin \
    firefox systemsettings kde-gtk-config \
    sddm plasma-wayland-session

~400MB, choose your own apps

Full Plasma (kde-apps)

sudo pacman -S plasma kde-applications \
    sddm plasma-wayland-session

~2GB, includes all KDE apps

3.2 Enable SDDM & Services

sudo systemctl enable sddm
sudo systemctl enable NetworkManager
sudo systemctl enable bluetooth  # if needed

# For Wayland session (recommended)
# Ensure your user is in video group
sudo usermod -aG video $USER

3.3 Essential Plasma Packages for Ricing

# Core ricing tools
sudo pacman -S --needed \
    kvantum-qt5 kvantum-qt6 \
    kde-gtk-config \
    qt5ct qt6ct \
    plasma-browser-integration \
    xdg-desktop-portal-kde \
    kdeconnect

# AUR packages for advanced ricing
yay -S --needed \
    latte-dock-git \
    krohnkite-git \
    bismuth-git \
    plasma5-theme-nordic \
    plasma6-theme-catppuccin-git \
    whitesur-kde-theme-git \
    tokyo-night-kde-theme-git \
    gruvbox-kde-theme-git

4. Graphics Drivers

sudo pacman -S mesa libva-mesa-driver \
    mesa-vdpau vulkan-intel
# For older Intel
# sudo pacman -S intel-media-driver
sudo pacman -S mesa libva-mesa-driver \
    mesa-vdpau vulkan-radeon \
    lib32-mesa lib32-vulkan-radeon \
    lib32-libva-mesa-driver
# Proprietary (recommended for gaming/CUDA)
sudo pacman -S nvidia nvidia-utils \
    lib32-nvidia-utils nvidia-settings \
    vulkan-icd-loader lib32-vulkan-icd-loader

# Or open-source (nouveau - limited features)
# sudo pacman -S xf86-video-nouveau mesa

# Add kernel modules for mkinitcpio
sudo sed -i 's/MODULES=()/MODULES=(nvidia nvidia_modeset nvidia_uvm nvidia_drm)/' /etc/mkinitcpio.conf
sudo mkinitcpio -P

# Enable DRM KMS
echo "options nvidia_drm modeset=1 fbdev=1" | sudo tee /etc/modprobe.d/nvidia.conf
# For Optimus laptops (Intel + NVIDIA)
sudo pacman -S nvidia-prime optimus-manager

# Or use envycontrol (AUR)
yay -S envycontrol

# Switch GPU:
# envycontrol -s integrated    # Intel only
# envycontrol -s hybrid        # Dynamic switching
# envycontrol -s nvidia        # NVIDIA only

5. Audio Setup (PipeWire)

sudo pacman -S pipewire pipewire-pulse \
    pipewire-alsa pipewire-jack \
    wireplumber libwireplumber \
    pavucontrol helvum  # GUI mixers

# Enable user services
systemctl --user enable pipewire pipewire-pulse wireplumber

6. Theming Basics

6.1 Global Themes

Install themes via KDE Store (Get New Global Themes) or manually:

# Manual theme installation
mkdir -p ~/.local/share/plasma/look-and-feel
# Extract theme there

# Or use plasmastyle (AUR) for CLI theming
yay -S plasmastyle
plasmastyle apply Nordic

6.2 Color Schemes

# Popular color schemes (install via KDE Store or AUR)
# Nordic, Catppuccin, Gruvbox, Tokyo Night, Dracula, Rose Pine

# Apply via CLI
plasma-apply-colorscheme Nordic
# Or for specific scheme file
plasma-apply-colorscheme ~/.local/share/color-schemes/Catppuccin-Mocha.colors

6.3 Application Style & Fonts

# Fonts
sudo pacman -S noto-fonts noto-fonts-emoji \
    ttf-jetbrains-mono ttf-firacode-nerd \
    ttf-cascadia-code-nerd

# Qt theme config (for GTK apps)
kwriteconfig6 --file kdeglobals --group General --key Font "Noto Sans,10,-1,5,50,0,0,0,0,0"
kwriteconfig6 --file kdeglobals --group General --key Fixed "JetBrains Mono,10,-1,5,50,0,0,0,0,0"
kwriteconfig6 --file kdeglobals --group General --key MenuFont "Noto Sans,10,-1,5,50,0,0,0,0,0"

# Force Qt apps to use KDE style
echo "export QT_QPA_PLATFORMTHEME=kde" >> ~/.config/environment.d/envvars.conf

6.4 Kvantum (Qt Theming Engine)

# Install Kvantum themes
yay -S kvantum-theme-nordic kvantum-theme-catppuccin-git \
    kvantum-theme-gruvbox-git kvantum-theme-tokyonight-git

# Apply theme
kvantummanager --set Nordic
# Or GUI: kvantummanager

8. Dotfile Management & Backup

8.1 Bare Git Repository (Recommended)

# Initialize bare repo
git init --bare $HOME/.dotfiles
alias dotfiles='/usr/bin/git --git-dir=$HOME/.dotfiles/ --work-tree=$HOME'
dotfiles config --local status.showUntrackedFiles no

# Add configs
dotfiles add .config/kdeglobals .config/kwinrc .config/plasma-org.kde.plasma.desktop-appletsrc ...
dotfiles commit -m "Initial KDE config"
dotfiles remote add origin git@github.com:user/dotfiles.git
dotfiles push -u origin main

8.2 Stow (Symlink Manager)

sudo pacman -S stow

# Structure: ~/dotfiles/.config/kdeglobals → ~/.config/kdeglobals
cd ~/dotfiles
stow .config .local

8.3 Home Manager (Nix-style on Arch)

# Install via Nix (multi-user)
sh <(curl -L https://nixos.org/nix/install) --daemon

# Then use Home Manager for declarative config
# See NixOS guide for Home Manager details

8.4 Backup Critical Configs

#!/bin/bash
# backup-kde.sh
BACKUP_DIR="$HOME/backups/kde-$(date +%Y%m%d)"
mkdir -p "$BACKUP_DIR"

cp -r ~/.config/kde* "$BACKUP_DIR/"
cp -r ~/.config/plasma* "$BACKUP_DIR/"
cp -r ~/.config/kwin* "$BACKUP_DIR/"
cp -r ~/.local/share/plasma "$BACKUP_DIR/"
cp -r ~/.local/share/color-schemes "$BACKUP_DIR/"
cp -r ~/.local/share/kwin "$BACKUP_DIR/"

echo "Backed up to $BACKUP_DIR"
